# Сборка
docker build -t cluster:latest -f Cluster/Dockerfile .

# Экспорт созданного образа
docker save -o cluster-app.tar cluster:latest

# Перенос на менеджер
scp cluster-app.tar ubuntu@192.168.10.12:/home/ubuntu/

# Выполнить на виртуалках
sudo docker load -i /home/ubuntu/cluster-app.tar

# Проверка
sudo docker stack ps my-mvc-app
docker service ps mvc

# Перебалансировка
sudo docker service update --force --image cluster:latest mvc

# Смена мастера
sudo patronictl -c /etc/patroni/config.yml switchover \
  --master vm-db \
  --candidate vm-app \
  --force

# Роли серверов
sudo patronictl -c /etc/patroni/config.yml list

# Нчало задержки
docker service logs -f --since 0s --timestamps my-mvc-app_crm_web 2>&1 | grep --line-buffered -F "TEST: transaction is ready but not committed"
